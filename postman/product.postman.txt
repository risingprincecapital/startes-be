# EXAMPLE WORKFLOW
# =============================================================================

# Step 1: Create a new product and capture the ID
RESPONSE=$(curl -X POST http://localhost:3000/api/products \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "productName": "EIN Application Service",
    "description": "Federal Employer Identification Number application service",
    "price": 99,
    "processType": "standard",
    "departmentType": "federal",
    "timeToComplete": 3,
    "productType": "onetime",
    "category": "Tax Services",
    "subCategory": "EIN"
  }')

# Extract product ID from response (requires jq)
NEW_PRODUCT_ID=$(echo $RESPONSE | jq -r '.product._id')
echo "Created product with ID: $NEW_PRODUCT_ID"

# Step 2: Get the product details
curl -X GET http://localhost:3000/api/products/$NEW_PRODUCT_ID \
  -H "Authorization: Bearer $TOKEN"

# Step 3: Update the product
curl -X PUT http://localhost:3000/api/products/$NEW_PRODUCT_ID \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "price": 129,
    "description": "Fast EIN application service with expert support"
  }'

# Step 4: Get all products to see the updated list
curl -X GET "http://localhost:3000/api/products?limit=10&page=1" \
  -H "Authorization: Bearer $TOKEN"
